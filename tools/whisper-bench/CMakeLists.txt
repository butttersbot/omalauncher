# The desktop bench for the phone's Whisper (see bench.cpp). Build:
#   cmake -S tools/whisper-bench -B /tmp/whisper-bench -DCMAKE_BUILD_TYPE=Release && cmake --build /tmp/whisper-bench -j
cmake_minimum_required(VERSION 3.22)
project(omawhisper_bench C CXX)
set(CPP ${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
add_subdirectory(${CPP}/whisper.cpp/ggml ggml)
add_library(whisper STATIC ${CPP}/whisper.cpp/src/whisper.cpp ${CPP}/whisper.cpp/src/aneforge/whisper-aneforge.cpp)
target_include_directories(whisper PUBLIC ${CPP}/whisper.cpp/include ${CPP}/whisper.cpp/src)
target_compile_features(whisper PUBLIC cxx_std_17)
target_compile_definitions(whisper PRIVATE WHISPER_VERSION="1.9.5")
target_link_libraries(whisper PUBLIC ggml)
add_executable(bench bench.cpp)
target_include_directories(bench PRIVATE ${CPP})
target_link_libraries(bench whisper)
